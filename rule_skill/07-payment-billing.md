// src/application/services/invoice/auto-generate.usecase.ts
export class AutoGenerateInvoiceUseCase {
  constructor(
    private invoiceRepo: IInvoiceRepository,
    private attendanceRepo: IAttendanceRepository,
    private enrollmentRepo: IEnrollmentRepository,
    private notificationService: INotificationService,
    private taxCalculator: ITaxCalculator
  ) {}

  async execute(dto: AutoGenerateDTO): Promise<Result<Invoice[]>> {
    const { teacherId, classId, month, year } = dto;
    
    const enrollments = await this.enrollmentRepo.findActiveByClass(classId);
    const schedule = await this.scheduleRepo.findByClass(classId);
    
    const sessionsInMonth = this.calculateSessions(schedule, month, year);
    
    const invoices: Invoice[] = [];
    
    for (const enrollment of enrollments) {
      const attendance = await this.attendanceRepo.findByStudentAndMonth(
        enrollment.studentId, classId, month, year
      );
      
      const presentCount = attendance.filter(a => a.status === 'present').length;
      const lateCount = attendance.filter(a => a.status === 'late').length;
      
      const effectiveSessions = presentCount + (lateCount * 0.5);
      const unitPrice = enrollment.customFee ?? class.feePerSession;
      const subtotal = effectiveSessions * unitPrice;
      
      const taxRate = await this.taxCalculator.getRate(teacherId);
      const taxAmount = subtotal * taxRate;
      
      const invoice = Invoice.create({
        teacherId,
        studentId: enrollment.studentId,
        classId,
        periodStart: startOfMonth,
        periodEnd: endOfMonth,
        sessionsCount: effectiveSessions,
        unitPrice,
        subtotal,
        taxRate,
        totalAmount: subtotal + taxAmount,
        lineItems: [{
          description: `Hoc phi thang ${month}/${year}`,
          quantity: effectiveSessions,
          unitPrice,
          amount: subtotal
        }]
      });
      
      if (invoice.isSuccess()) {
        invoices.push(invoice.getValue());
      }
    }
    
    await this.invoiceRepo.saveMany(invoices);
    
    for (const invoice of invoices) {
      await this.notificationService.sendInvoiceNotification(invoice);
    }
    
    return Result.ok(invoices);
  }
}

// Khi nhan webhook tu payment gateway
export class ProcessPaymentWebhookUseCase {
  async execute(payload: WebhookPayload): Promise<Result<void>> {
    const gateway = this.gatewayFactory.get(payload.gateway);
    const verification = gateway.verifyWebhook(payload.data, payload.signature);
    if (!verification.isSuccess()) return Result.fail(verification.getError());
    
    const data = verification.getValue();
    
    const existing = await this.transactionRepo.findByGatewayRef(data.transactionRef);
    if (existing) return Result.ok(undefined);
    
    const invoice = await this.invoiceRepo.findById(data.invoiceId);
    if (!invoice) return Result.fail(new DomainError("Invoice not found"));
    
    const payment = Payment.create({
      invoiceId: invoice.id,
      amount: data.amount,
      method: data.method,
      gatewayRef: data.transactionRef,
      paidAt: new Date()
    });
    
    const result = invoice.applyPayment(payment);
    if (!result.isSuccess()) return result;
    
    await this.invoiceRepo.save(invoice);
    await this.transactionRepo.save(payment);
    
    await this.notificationService.notifyPaymentSuccess(invoice);
    await this.analyticsService.recordRevenue(invoice.teacherId, data.amount);
    
    return Result.ok(undefined);
  }
}

// src/application/services/tax/generate-tax-report.usecase.ts
export class GenerateTaxReportUseCase {
  async execute(teacherId: string, year: number, quarter?: number): Promise<TaxReport> {
    const startDate = quarter 
      ? startOfQuarter(new Date(year, (quarter - 1) * 3, 1))
      : startOfYear(new Date(year, 0, 1));
    const endDate = quarter
      ? endOfQuarter(startDate)
      : endOfYear(startDate);
    
    const invoices = await this.invoiceRepo.findByTeacherAndDateRange(
      teacherId, startDate, endDate
    );
    
    const totalRevenue = invoices
      .filter(i => i.status === 'paid')
      .reduce((sum, i) => sum + i.totalAmount, 0);
    
    const taxConfig = await this.taxConfigRepo.findByTeacher(teacherId);
    const taxAmount = this.calculateTax(totalRevenue, taxConfig);
    
    return TaxReport.create({
      teacherId,
      year,
      quarter,
      totalRevenue,
      taxAmount,
      reportData: this.formatReportData(invoices)
    });
  }
}
